Add detection for when an expert's Stripe

1. Add a webhook handler for account status 
   changes:

case "account.updated": {
  const account = event.data.object;
  
  // Find the expert by their Connect account ID
  const expert = await db.select()
    .from(localExpertForms)
    .where(eq(
      localExpertForms.stripeConnectAccountId, 
      account.id
    ));

  if (expert.length === 0) break;

  const isRestricted = 
    account.requirements?.disabled_reason !== null;
  const chargesEnabled = account.charges_enabled;
  const payoutsEnabled = account.payouts_enabled;

  let newStatus = "complete";
  if (isRestricted || !chargesEnabled || 
      !payoutsEnabled) {
    newStatus = "restricted";
    
    // Alert admin immediately — this expert
    // may still be receiving leads but can't
    // be paid
    await db.insert(adminNotifications).values({
      type: "expert_stripe_restricted",
      message: 
        `Expert ${expert[0].id} Stripe account ` +
        `restricted: ${account.requirements
          ?.disabled_reason || "unknown reason"}`,
      reason: "stripe_account_restricted"
    });

    console.error(
      `[STRIPE RESTRICTION] Expert ${expert[0].id}` +
      ` account restricted. Reason: ${account
        .requirements?.disabled_reason}`
    );
  }

  await db.update(localExpertForms)
    .set({ stripeConnectStatus: newStatus })
    .where(eq(localExpertForms.id, expert[0].id));

  break;
}
