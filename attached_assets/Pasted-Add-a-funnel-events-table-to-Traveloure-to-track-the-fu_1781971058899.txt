export const funnelStageEnum = pgEnum(
  "funnel_stage", [
  "T0_LANDING",
  "T1_ACCOUNT_CREATED", 
  "T2_TRIP_CREATED",
  "T3_ITINERARY_GENERATED",
  "T4_CART_POPULATED",
  "T5_EXPERT_REQUESTED",
  "T6_REVENUE",
  "T7_VIRAL_SHARE"
]);

export const funnelEvents = pgTable(
  "funnel_events", {
  id: serial("id").primaryKey(),
  userId: integer("user_id")
    .references(() => users.id),
  sessionId: text("session_id"),
  eventType: text("event_type").notNull(),
  funnelStage: funnelStageEnum(
    "funnel_stage").notNull(),
  tripId: integer("trip_id"),
  expertRequestId: integer("expert_request_id"),
  bookingId: integer("booking_id"),
  source: text("source"),
  refToken: text("ref_token"),
  eventData: jsonb("event_data"),
  createdAt: timestamp("created_at").defaultNow()
});

// Add index for fast funnel queries
funnelEventsUserIdx: index(
  "funnel_events_user_id_idx")

2. Create server/utils/funnelTracker.ts:

import { db } from "../db";
import { funnelEvents } from "../../shared/schema";

export async function trackFunnelEvent(params: {
  userId?: number;
  sessionId?: string;
  eventType: string;
  funnelStage: string;
  tripId?: number;
  expertRequestId?: number;
  bookingId?: number;
  source?: string;
  refToken?: string;
  eventData?: Record<string, unknown>;
}) {
  try {
    await db.insert(funnelEvents).values(params);
  } catch (err) {
    // Never block the main flow for analytics
    console.error("[FUNNEL TRACK ERROR]", err);
  }
}
