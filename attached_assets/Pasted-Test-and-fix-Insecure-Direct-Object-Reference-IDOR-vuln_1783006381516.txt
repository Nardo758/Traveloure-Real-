STEP 2 — Add ownership middleware where missing:

Create server/middleware/ownershipGuard.ts:

export function requireOwnership(
  getResourceUserId: (req: any) => Promise<number | null>
) {
  return async (req: any, res: any, next: any) => {
    try {
      const resourceUserId = await getResourceUserId(req);
      
      if (resourceUserId === null) {
        return res.status(404).json({ 
          message: "Not found" 
        });
      }
      
      if (resourceUserId !== req.user.id && 
          req.user.role !== "admin") {
        console.warn(
          `[IDOR ATTEMPT] User ${req.user.id} ` +
          `tried to access resource owned by ` +
          `${resourceUserId} at ${req.path}`
        );
        return res.status(403).json({ 
          message: "Access denied" 
        });
      }
      
      next();
    } catch (err) {
      res.status(500).json({ message: "Error" });
    }
  };
}
