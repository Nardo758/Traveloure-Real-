const slowQueryLog: Array<{
  label: string;
  duration: number;
  threshold: number;
  userRole: string;
  timestamp: string;
}> = [];

const MAX_LOG_SIZE = 100;

// Add to withQueryTimer after console.warn:
slowQueryLog.unshift({
  label,
  duration,
  threshold: THRESHOLD,
  userRole: userRole || "unknown",
  timestamp: new Date().toISOString()
});
if (slowQueryLog.length > MAX_LOG_SIZE) {
  slowQueryLog.pop();
}

export function getSlowQueryLog() {
  return [...slowQueryLog];
}

export function clearSlowQueryLog() {
  slowQueryLog.length = 0;
}

2. Add admin endpoint in routes.ts:

import { 
  getSlowQueryLog, 
  clearSlowQueryLog 
} from "./utils/queryTimer";

// GET slow queries — admin only
app.get("/api/admin/slow-queries", 
  requireAdmin, // existing admin auth middleware
  (req, res) => {
    res.json({
      count: getSlowQueryLog().length,
      threshold: process.env
        .SLOW_QUERY_THRESHOLD_MS || 500,
      queries: getSlowQueryLog()
    });
  }
);

// DELETE to clear log — admin only  
app.delete("/api/admin/slow-queries",
  requireAdmin,
  (req, res) => {
    clearSlowQueryLog();
    res.json({ message: "Slow query log cleared" });
  }
);
