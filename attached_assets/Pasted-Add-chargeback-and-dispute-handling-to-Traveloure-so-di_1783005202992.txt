1. Add Stripe webhook handler for disputes:

In your webhook processing function, add:

case "charge.dispute.created": {
  const dispute = event.data.object;
  const charge = await stripe.charges.retrieve(
    dispute.charge as string
  );
  const bookingId = charge.metadata?.bookingId;

  if (bookingId) {
    // Mark booking as disputed
    await db.update(bookings)
      .set({ 
        status: "disputed",
        disputeId: dispute.id,
        disputeReason: dispute.reason
      })
      .where(eq(bookings.id, parseInt(bookingId)));

    // Alert admin immediately
    await db.insert(adminNotifications).values({
      type: "dispute_created",
      message: 
        `Chargeback filed for booking ${bookingId}` +
        ` — reason: ${dispute.reason}`,
      reason: dispute.reason
    });

    console.warn(
      `[DISPUTE] Booking ${bookingId} disputed.` +
      ` Reason: ${dispute.reason}`
    );

    // If expert already received payout for this
    // booking, flag for manual review — do NOT
    // automatically claw back funds
    const expertRequest = await db.select()
      .from(expertRequests)
      .where(eq(expertRequests.tripId, 
        booking.tripId));
    
    if (expertRequest[0]?.status === "completed") {
      await db.insert(adminNotifications).values({
        type: "dispute_after_payout",
        message: 
          `URGENT: Booking ${bookingId} disputed ` +
          `but expert may have already been paid. ` +
          `Manual review required.`,
        reason: "payout_clawback_review"
      });
    }
  }
  break;
}
