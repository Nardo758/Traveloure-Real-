import { Router } from "express";
import { db } from "../db";
import { sql } from "drizzle-orm";

const router = Router();

router.get("/ready", async (req, res) => {
  const checks: Record<string, {
    status: "ok" | "fail" | "warn";
    message: string;
  }> = {};

  // CHECK 1: Database connectivity
  try {
    await db.execute(sql`SELECT 1`);
    checks.database = { 
      status: "ok", 
      message: "Connected" 
    };
  } catch (err) {
    checks.database = { 
      status: "fail", 
      message: "Cannot reach database" 
    };
  }

  // CHECK 2: AI service (XAI key present)
  const xaiKeyPresent = Boolean(
    process.env.XAI_API_KEY
  );
  checks.ai_xai = {
    status: xaiKeyPresent ? "ok" : "fail",
    message: xaiKeyPresent 
      ? "XAI_API_KEY present" 
      : "XAI_API_KEY missing — " +
        "AI generation will fail. " +
        "Set in GitHub Secrets → Actions."
  };

  // CHECK 3: Anthropic/Claude key present
  const claudeKeyPresent = Boolean(
    process.env.ANTHROPIC_API_KEY
  );
  checks.ai_claude = {
    status: claudeKeyPresent ? "ok" : "warn",
    message: claudeKeyPresent
      ? "ANTHROPIC_API_KEY present"
      : "ANTHROPIC_API_KEY missing — " +
        "chat/optimization will degrade"
  };

  // CHECK 4: Stripe configured
  const stripeKeyPresent = Boolean(
    process.env.STRIPE_SECRET_KEY
  );
  checks.stripe = {
    status: stripeKeyPresent ? "ok" : "fail",
    message: stripeKeyPresent
      ? "STRIPE_SECRET_KEY present"
      : "STRIPE_SECRET_KEY missing — " +
        "payments will fail"
  };

  // CHECK 5: Resend email configured
  const resendKeyPresent = Boolean(
    process.env.RESEND_API_KEY
  );
  checks.email = {
    status: resendKeyPresent ? "ok" : "warn",
    message: resendKeyPresent
      ? "RESEND_API_KEY present"
      : "RESEND_API_KEY missing — " +
        "digest emails will not send"
  };

  // CHECK 6: Webhook secret configured
  const webhookSecretPresent = Boolean(
    process.env.STRIPE_WEBHOOK_SECRET
  );
  checks.webhook = {
    status: webhookSecretPresent ? "ok" : "fail",
    message: webhookSecretPresent
      ? "STRIPE_WEBHOOK_SECRET present"
      : "STRIPE_WEBHOOK_SECRET missing — " +
        "webhooks will be rejected"
  };

  // Overall status
  const hasFail = Object.values(checks)
    .some(c => c.status === "fail");
  const hasWarn = Object.values(checks)
    .some(c => c.status === "warn");

  const overall = hasFail 
    ? "fail" 
    : hasWarn 
    ? "warn" 
    : "ok";

  const httpStatus = hasFail ? 503 : 200;

  return res.status(httpStatus).json({
    status: overall,
    timestamp: new Date().toISOString(),
    environment: process.env.NODE_ENV || 
      "unknown",
    checks
  });
});

export default router;

3. Update the CI workflow file
   (.github/workflows/*.yml):

Find the server-check step that currently
just pings /api/health or waits for boot.

Replace or add after it:

- name: Verify service readiness
  run: |
    echo "Checking /api/ready..."
    READY=$(curl -s -o /tmp/ready.json \
      -w "%{http_code}" \
      ${{ env.BASE_URL }}/api/ready)
    cat /tmp/ready.json | jq .
    if [ "$READY" != "200" ]; then
      echo "❌ Readiness check failed:"
      cat /tmp/ready.json | jq '.checks | 
        to_entries[] | 
        select(.value.status == "fail") | 
        .key + ": " + .value.message'
      exit 1
    fi
    echo "✅ All critical services ready"
