Two commission models exist and conflict:

1. Create server/utils/commissionCalculator.ts:

export enum BookingType {
  EXPERT_SESSION = "EXPERT_SESSION",
  PROVIDER_BOOKING = "PROVIDER_BOOKING", 
  EXPERIENCE_CART = "EXPERIENCE_CART",
  CREDIT_PURCHASE = "CREDIT_PURCHASE"
}

export interface CommissionResult {
  platformFee: number;
  expertPayout?: number;
  providerPayout?: number;
  bookingType: BookingType;
}

export function calculateCommission(
  subtotal: number,
  bookingType: BookingType,
  options?: {
    isNewExpert?: boolean;
    providerTier?: 1 | 2 | 3 | 4;
  }
): CommissionResult {
  switch (bookingType) {
    
    case BookingType.EXPERT_SESSION:
      // New expert: 85/15, established: 75/25
      const expertRate = options?.isNewExpert 
        ? 0.85 : 0.75;
      const platformRate = 1 - expertRate;
      return {
        platformFee: subtotal * platformRate,
        expertPayout: subtotal * expertRate,
        bookingType
      };

    case BookingType.PROVIDER_BOOKING:
      // Tier 1=12%, 2=8%, 3=6%, 4=4%
      const tierRates = { 1:0.12, 2:0.08, 
                          3:0.06, 4:0.04 };
      const tier = options?.providerTier || 1;
      const commission = tierRates[tier];
      return {
        platformFee: subtotal * commission,
        providerPayout: subtotal * (1 - commission),
        bookingType
      };

    case BookingType.EXPERIENCE_CART:
      // 30% platform fee (current checkout code)
      return {
        platformFee: subtotal * 0.30,
        bookingType
      };

    case BookingType.CREDIT_PURCHASE:
      // Full amount = deferred revenue
      return {
        platformFee: subtotal,
        bookingType
      };
  }
}
