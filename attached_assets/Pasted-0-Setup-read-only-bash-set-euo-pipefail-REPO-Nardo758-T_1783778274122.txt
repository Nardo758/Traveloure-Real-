4) Lockfile exact diff + replit.local grep in branch lockfiles
bash
for f in package-lock.json pnpm-lock.yaml yarn.lock; do
  if git cat-file -e origin/${BRANCH}:$f 2>/dev/null; then
    echo "=== $f diff vs main ==="
    git diff origin/${BASE}...origin/${BRANCH} -- $f || true
    echo "=== grep replit.local in $f @ branch ==="
    git grep -n "replit.local" origin/${BRANCH} -- $f || true
  fi
done
5) Exact patch for intended file
bash
# if exactly one changed file, print patch
ONE_FILE=$(git diff --name-only origin/${BASE}...origin/${BRANCH} | wc -l | tr -d ' ')
if [ "$ONE_FILE" = "1" ]; then
  FILE=$(git diff --name-only origin/${BASE}...origin/${BRANCH})
  echo "=== intended file patch: $FILE ==="
  git diff origin/${BASE}...origin/${BRANCH} -- "$FILE"
else
  echo "Branch changes more than one file; contamination likely."
fi

Then classify each gate as red-on-main or not:

bash
echo -e "gate\tred_on_pr\tred_on_main\tclassification"
while IFS= read -r gate; do
  pr=$(echo "$PR_JSON" | jq -r --arg g "$gate" '
    [ .statusCheckRollup[]? | select(.name==$g) | (.conclusion // .status // "UNKNOWN") ][0] // "NOT_FOUND"
  ')
  red_pr="no"; [[ "$pr" =~ ^(FAILURE|TIMED_OUT|CANCELLED|ACTION_REQUIRED|failure|timed_out|cancelled)$ ]] && red_pr="yes"

  # if any recent main job with same name failed, mark red_on_main=yes
  red_main=$(awk -F'\t' -v g="$gate" '
    $1==g && ($2=="failure" || $2=="timed_out" || $2=="cancelled" || $2=="FAILURE" || $2=="TIMED_OUT" || $2=="CANCELLED"){print "yes"; found=1}
    END{if(!found) print "no"}
  ' /tmp/main_jobs.tsv)

  cls="introduced"
  if [ "$red_pr" = "yes" ] && [ "$red_main" = "yes" ]; then cls="inherited baseline"; fi
  if [ "$red_pr" = "no" ]; then cls="not red on PR"; fi

  printf "%s\t%s\t%s\t%s\n" "$gate" "$red_pr" "$red_main" "$cls"
done < /tmp/gates.txt
